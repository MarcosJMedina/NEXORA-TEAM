  <button id="ia-chat-toggle" style="background-color: #0f6cbf; color: white; border: none; border-radius: 50%; width: 60px; height: 60px; cursor: pointer; box-shadow: 0 4px 8px rgba(0,0,0,0.2); font-size: 24px;">💬</button>
  
  <div id="ia-chat-box" style="display: none; position: absolute; bottom: 70px; right: 0; width: 350px; height: 450px; background: white; border-radius: 12px; box-shadow: 0 5px 15px rgba(0,0,0,0.3); flex-direction: column; overflow: hidden; border: 1px solid #ccc;">
    <div style="background: #0f6cbf; color: white; padding: 12px 16px; font-weight: bold; display: flex; justify-content: space-between; align-items: center;">
      <span>Asistente de IA (Química)</span>
      <button id="ia-chat-close" style="background: none; border: none; color: white; font-size: 18px; cursor: pointer;">✕</button>
    </div>
    
    <div id="ia-chat-messages" style="flex: 1; padding: 12px; overflow-y: auto; display: flex; flex-direction: column; gap: 8px;">
      <div style="background: #f1f1f1; padding: 8px 12px; border-radius: 8px; max-width: 80%; align-self: flex-start;">
        Hola 👋, soy tu tutor de IA. ¿Tienes alguna duda sobre las fechas o el material del curso?
      </div>
    </div>
    
    <div style="padding: 10px; border-top: 1px solid #eee; display: flex; gap: 6px;">
      <input type="text" id="ia-chat-input" placeholder="Escribe tu duda académica..." style="flex: 1; border: 1px solid #ccc; border-radius: 20px; padding: 8px 12px; outline: none; font-size: 14px;">
      <button id="ia-chat-send" style="background: #0f6cbf; color: white; border: none; border-radius: 20px; padding: 8px 16px; cursor: pointer; font-weight: bold;">Enviar</button>
    </div>
  </div>
</div>

<script>
document.addEventListener('DOMContentLoaded', function () {
    const toggleBtn = document.getElementById('ia-chat-toggle');
    const closeBtn = document.getElementById('ia-chat-close');
    const chatBox = document.getElementById('ia-chat-box');
    const sendBtn = document.getElementById('ia-chat-send');
    const input = document.getElementById('ia-chat-input');
    const messagesContainer = document.getElementById('ia-chat-messages');

    if (!toggleBtn || !chatBox) return;

    toggleBtn.addEventListener('click', () => {
        chatBox.style.display = chatBox.style.display === 'none' ? 'flex' : 'none';
    });

    closeBtn.addEventListener('click', () => {
        chatBox.style.display = 'none';
    });

    async function sendMessage() {
        const text = input.value.trim();
        if (!text) return;

        const userMsg = document.createElement('div');
        userMsg.style.cssText = 'background: #0f6cbf; color: white; padding: 8px 12px; border-radius: 8px; max-width: 80%; align-self: flex-end;';
        userMsg.textContent = text;
        messagesContainer.appendChild(userMsg);
        input.value = '';
        messagesContainer.scrollTop = messagesContainer.scrollHeight;

        const userId = (typeof M !== 'undefined' && M.cfg && M.cfg.userId) ? M.cfg.userId : 0;
        const courseId = (typeof M !== 'undefined' && M.cfg && M.cfg.courseId) ? M.cfg.courseId : 1;

        try {
            const response = await fetch('http://127.0.0.1:8000/api/chat', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    user_id: parseInt(userId),
                    course_id: parseInt(courseId),
                    message: text
                })
            });

            if (!response.ok) throw new Error('Error en la API backend');

            const data = await response.json();
            const botMsg = document.createElement('div');
            botMsg.style.cssText = 'background: #f1f1f1; color: #333; padding: 8px 12px; border-radius: 8px; max-width: 80%; align-self: flex-start;';
            botMsg.textContent = data.response;
            messagesContainer.appendChild(botMsg);
        } catch (err) {
            const errorMsg = document.createElement('div');
            errorMsg.style.cssText = 'background: #ffebee; color: #c62828; padding: 8px 12px; border-radius: 8px; max-width: 80%; align-self: flex-start; font-size: 12px;';
            errorMsg.textContent = 'Acceso denegado o error de conexión con el servidor.';
            messagesContainer.appendChild(errorMsg);
        }
        messagesContainer.scrollTop = messagesContainer.scrollHeight;
    }

    sendBtn.addEventListener('click', sendMessage);
    input.addEventListener('keypress', (e) => {
        if (e.key === 'Enter') sendMessage();
    });
});
</script>
